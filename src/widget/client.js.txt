(function () {
  var SQ = 90, MARGIN = 40;
  var NS = 'http://www.w3.org/2000/svg';
  var data = null, idx = 0, flipped = false;
  var svg = document.getElementById('board-svg');
  var hlLayer = document.createElementNS(NS, 'g');
  var pieceLayer = document.createElementNS(NS, 'g');
  svg.appendChild(hlLayer);
  svg.appendChild(pieceLayer);
  var movesEl = document.getElementById('moves');
  var commentEl = document.getElementById('comment');
  var statusEl = document.getElementById('status');
  var btn = {
    first: document.getElementById('b-first'),
    prev: document.getElementById('b-prev'),
    next: document.getElementById('b-next'),
    last: document.getElementById('b-last'),
  };

  function xy(i) {
    var col = i % 8, row = Math.floor(i / 8);
    return [MARGIN + (flipped ? 7 - col : col) * SQ, MARGIN + (flipped ? 7 - row : row) * SQ];
  }

  function fenAt(n) { return n === 0 ? data.startFen : data.plies[n - 1].fen; }

  function drawBoard() {
    pieceLayer.textContent = '';
    hlLayer.textContent = '';
    var placement = fenAt(idx).split(' ')[0].split('/');
    for (var r = 0; r < 8; r++) {
      var c = 0;
      for (var k = 0; k < placement[r].length; k++) {
        var ch = placement[r][k];
        if (ch >= '1' && ch <= '8') { c += +ch; continue; }
        var p = xy(r * 8 + c);
        var use = document.createElementNS(NS, 'use');
        use.setAttribute('href', '#' + (ch === ch.toUpperCase() ? 'w' : 'b') + ch.toUpperCase());
        use.setAttribute('x', p[0]); use.setAttribute('y', p[1]);
        use.setAttribute('width', SQ); use.setAttribute('height', SQ);
        pieceLayer.appendChild(use);
        c++;
      }
    }
    if (idx > 0) {
      var ply = data.plies[idx - 1];
      [ply.from, ply.to].forEach(function (sq) {
        var q = xy(sq);
        var rect = document.createElementNS(NS, 'rect');
        rect.setAttribute('x', q[0]); rect.setAttribute('y', q[1]);
        rect.setAttribute('width', SQ); rect.setAttribute('height', SQ);
        rect.setAttribute('fill', '#f6e84a'); rect.setAttribute('fill-opacity', '0.5');
        hlLayer.appendChild(rect);
      });
    }
  }

  function buildMoves() {
    movesEl.textContent = '';
    var row = null;
    data.plies.forEach(function (ply, i) {
      if (ply.color === 'w' || !row) {
        row = document.createElement('div');
        row.className = 'row';
        var num = document.createElement('span');
        num.className = 'num';
        num.textContent = ply.number + (ply.color === 'w' ? '.' : '...');
        row.appendChild(num);
        movesEl.appendChild(row);
        if (ply.color === 'b') {
          var gap = document.createElement('span'); gap.className = 'gap'; row.appendChild(gap);
        }
      }
      var b = document.createElement('button');
      b.type = 'button';
      b.className = 'mv';
      b.textContent = ply.san;
      b.setAttribute('data-i', i + 1);
      b.addEventListener('click', function () { go(i + 1); });
      row.appendChild(b);
    });
  }

  function go(n) {
    idx = Math.max(0, Math.min(data.plies.length, n));
    drawBoard();
    var active = movesEl.querySelector('.mv.on');
    if (active) active.classList.remove('on');
    if (idx > 0) {
      var cur = movesEl.querySelector('.mv[data-i="' + idx + '"]');
      if (cur) {
        cur.classList.add('on');
        var box = movesEl.getBoundingClientRect(), cb = cur.getBoundingClientRect();
        if (cb.top < box.top || cb.bottom > box.bottom) movesEl.scrollTop += cb.top - box.top - box.height / 2;
      }
    }
    commentEl.textContent = idx > 0 && data.plies[idx - 1].comment ? data.plies[idx - 1].comment : '';
    statusEl.textContent = idx + ' / ' + data.plies.length;
    btn.first.disabled = btn.prev.disabled = idx === 0;
    btn.next.disabled = btn.last.disabled = idx === data.plies.length;
  }

  function load(d) {
    data = d;
    flipped = false;
    var h = d.headers || {};
    document.getElementById('title').textContent = h.White || h.Black ? (h.White || '?') + ' vs ' + (h.Black || '?') : 'Chess';
    document.getElementById('subtitle').textContent = [h.Event, h.Date, d.result].filter(Boolean).join(' \u00b7 ');
    document.body.classList.toggle('static', d.plies.length === 0);
    buildMoves();
    go(typeof d.startPly === 'number' ? d.startPly : 0);
  }

  btn.first.addEventListener('click', function () { go(0); });
  btn.prev.addEventListener('click', function () { go(idx - 1); });
  btn.next.addEventListener('click', function () { go(idx + 1); });
  btn.last.addEventListener('click', function () { go(data.plies.length); });
  document.addEventListener('keydown', function (e) {
    if (!data) return;
    if (e.key === 'ArrowLeft') { go(idx - 1); e.preventDefault(); }
    else if (e.key === 'ArrowRight') { go(idx + 1); e.preventDefault(); }
    else if (e.key === 'Home') { go(0); e.preventDefault(); }
    else if (e.key === 'End') { go(data.plies.length); e.preventDefault(); }
  });

  // The game can arrive embedded (standalone page), in a tool result _meta.game (hidden from the model),
  // in structuredContent.game, via window.openai aliases, or via the MCP Apps postMessage bridge.
  function fromResult(r) {
    if (!r) return null;
    if (r._meta && r._meta.game) return r._meta.game;
    if (r.structuredContent && r.structuredContent.game) return r.structuredContent.game;
    return null;
  }

  function init() {
    var d = null;
    try { d = JSON.parse(document.getElementById('game-data').textContent); } catch (e) {}
    var o = window.openai;
    if (!d && o) d = (o.toolResponseMetadata && o.toolResponseMetadata.game) || (o.toolOutput && o.toolOutput.game) || null;
    if (d) load(d);
  }

  window.addEventListener('message', function (e) {
    var m = e.data;
    if (e.source !== window.parent || !m || m.jsonrpc !== '2.0') return;
    if (m.method === 'ui/notifications/tool-result') {
      var d = fromResult(m.params);
      if (d) load(d);
    }
  });
  window.addEventListener('openai:set_globals', init);
  window.chessmcpLoad = load;

  // MCP Apps handshake (best effort): tell the host we are ready so it sends the tool result.
  if (window.parent !== window) {
    try {
      window.parent.postMessage({ jsonrpc: '2.0', id: 1, method: 'ui/initialize', params: {
        appInfo: { name: 'chessmcp', version: '0.1.0' }, appCapabilities: {}, protocolVersion: '2025-06-18' } }, '*');
      window.parent.postMessage({ jsonrpc: '2.0', method: 'ui/notifications/initialized', params: {} }, '*');
    } catch (e) {}
  }
  init();
})();
